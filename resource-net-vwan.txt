
locals {
  vwan_friendly_name = "Virtual WAN"
  vwan_name          = "vwan-${var.prefix}"
  vwan_name_location = lower("${local.vwan_name}-${lower(var.location)}")
  vwan_random_suffix = substr(md5(local.vwan_name_location), 0, 6)
  vwan_name_hostname = lower(substr(replace("l${local.vwan_random_suffix}${local.vwan_name_location}", "-", ""), 0, 24))
  vwan_sku           = "Basic" ## Possible Values include: Basic and Standard. Defaults to Standard.
  vwan_hub_sku       = "Basic" ## Possible Values include: Basic and Standard. Defaults to Standard.
}

resource "azurerm_virtual_wan" "this" {
  name                = local.vwan_name
  resource_group_name = module.environment_resource_group.resource.name
  location            = azurerm_resource_group.environment.location

  disable_vpn_encryption            = false
  allow_branch_to_branch_traffic    = false
  office365_local_breakout_category = "All" ## Possible values include: Optimize, OptimizeAndAllow, All, None. Defaults to None.
  ## With Basic, the hubs are free!
  type = local.vwan_sku

  tags = { for key, value in azurerm_resource_group.environment.tags : key => value if lower(key) != "created" }
}

## $$ (unles vWAN is basic SKU) We need 1x vWAN Hub in each region (take a long time to deploy - 30+ minutes)
resource "azurerm_virtual_hub" "this" {
  name                = local.vwan_name
  resource_group_name = module.environment_resource_group.resource.name
  location            = azurerm_resource_group.environment.location

  virtual_wan_id = azurerm_virtual_wan.this.id

  ## address space
  address_prefix = "10.99.${local.regions[azurerm_resource_group.environment.location].location_number}.0/24"
  sku            = local.vwan_hub_sku ## Possible values are Basic and Standard. 

  hub_routing_preference = "ASPath" ## Possible values are ExpressRoute, ASPath and VpnGateway.

  tags = { for key, value in azurerm_resource_group.environment.tags : key => value if lower(key) != "created" }
}

resource "azurerm_vpn_gateway" "main" {
  count = local.local.vwan_hub_sku == "Standard" ? 1 : 0

  ## we only need one, since the Hubs are peered together (with Standard SKU, not Basic!)

  name                                  = "vpn-gateway-${var.environment}-${title(each.value.short_name)}"
  resource_group_name = module.environment_resource_group.resource.name
  location            = azurerm_resource_group.environment.location
  virtual_hub_id                        = azurerm_virtual_hub.this.id
  bgp_route_translation_for_nat_enabled = false
  routing_preference                    = "Microsoft Network"

  tags = { for key, value in azurerm_resource_group.environment.tags : key => value if lower(key) != "created" }
}

resource "azurerm_virtual_hub_connection" "vnet" {
  name                      = "link-VNET:${azurerm_virtual_network.this.name}-to-HUB:${azurerm_virtual_hub.this.name}"
  virtual_hub_id            = azurerm_virtual_hub.this.id
  remote_virtual_network_id = azurerm_virtual_network.this.id
}
