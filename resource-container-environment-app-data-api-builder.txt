module "dab_container_app" {
  source           = "Azure/avm-res-app-containerapp/azurerm"
  version          = "~> 0.9, < 1.0"
  enable_telemetry = var.enable_telemetry

  name                                  = "dab"
  resource_group_name                   = module.environment_resource_group.resource.name
  container_app_environment_resource_id = azurerm_container_app_environment.this.id
  revision_mode                         = "Single"

  managed_identities = {
    user_assigned_resource_ids = [azurerm_user_assigned_identity.environment.id]
  }

  ingress = {
    external_enabled = true
    target_port      = 80
    traffic_weight = [{
      percentage      = 100
      latest_revision = true
    }]
  }

  template = {
    volumes = [
      {
        name          = azurerm_container_app_environment_storage.logs.name
        storage_name  = azurerm_storage_account.this.name
        storage_type  = "AzureFile" ## NfsAzureFile
        mount_options = local.cifs_mount_options
      },
      {
        name          = azurerm_container_app_environment_storage.shared.name
        storage_name  = azurerm_storage_account.this.name
        storage_type  = "AzureFile" ## NfsAzureFile
        mount_options = local.cifs_mount_options
      }
    ]
    containers = [{
      name   = "dab"
      image  = "mcr.microsoft.com/azure-databases/data-api-builder:latest"
      cpu    = 0.5
      memory = "1Gi"
      env = [
        {
          name  = "AZURE_CLIENT_ID"
          value = azurerm_user_assigned_identity.environment.client_id
        },
        {
          name  = "AZURE_APP_CONFIGURATION_ID"
          value = module.appconfiguration.endpoint
        }
      ]
      volume_mounts = [
        {
          name = azurerm_container_app_environment_storage.logs.name
          path = "/mnt/logs"
        }
      ]
    }]
  }

  tags = { for key, value in module.environment_resource_group.resource.tags : key => value if lower(key) != "created" }

  depends_on = [
    azurerm_container_app_environment.this,
    azurerm_container_app_environment_storage.logs,
    azurerm_container_app_environment_storage.shared,
    azapi_update_resource.aca_env_otel,
    module.environment_resource_group
  ]
}

resource "azurerm_monitor_diagnostic_setting" "azurerm_container_app_dab" {
  name                       = "Metrics-${module.dab_container_app.resource.name}-${module.log_analytics_workspace.resource.name}-to-Azure-Monitor"
  target_resource_id         = module.dab_container_app.resource_id
  log_analytics_workspace_id = module.log_analytics_workspace.resource.resource_id

  enabled_metric {
    category = "AllMetrics"
  }
}
