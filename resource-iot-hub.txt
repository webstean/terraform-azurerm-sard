resource "azurerm_iothub" "this" {
  name                = local.iothub_name_hostname
  resource_group_name = module.environment_resource_group.resource.name
  location            = module.environment_resource_group.resource.location

  event_hub_partition_count = 2

  sku {
    ## Free (F), Basic (B) -$$ , Standard (S) -$$$
    name     = "${local.iot_hub_sku}${local.iot_hub_capacity}"
    capacity = local.iot_hub_capacity
  }

  network_rule_set {
    default_action = "Allow"
    ip_rule {
      name    = "default"
      action  = "Allow"
      ip_mask = "0.0.0.0/0"
    }
  }

  local_authentication_enabled  = (tobool(var.data_pii) || tobool(var.data_phi) || tobool(var.deploy_private_endpoints)) ? false : true
  public_network_access_enabled = (tobool(var.data_pii) || tobool(var.data_phi) || tobool(var.deploy_private_endpoints)) ? false : true

  ### endpoint types are:
  ### AzureIotHub.StorageContainer, AzureIotHub.ServiceBusQueue, AzureIotHub.ServiceBusTopic
  ### or AzureIotHub.EventHub

  cloud_to_device {
    max_delivery_count = 30
    default_ttl        = "PT1H"
    feedback {
      time_to_live       = "PT1H10M"
      max_delivery_count = 15
      lock_duration      = "PT30S"
    }
  }

  identity {
    type         = "SystemAssigned, UserAssigned"
    identity_ids = [azurerm_user_assigned_identity.environment.id]
  }

  tags = { for key, value in module.environment_resource_group.resource.tags : key => value if lower(key) != "created" }
}
