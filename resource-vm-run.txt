# authorize to storage blob using user assigned identity
resource "azurerm_virtual_machine_run_command" "run_setup_script" {
  for_each = toset([for i in range(local.vmss_number_of_instances) : format("%s_%d", local.vmms_name, i)])

  name               = "vm-setup-script"
  location           = module.environment_resource_group.resource.location
  virtual_machine_id = "${module.environment_resource_group.resource_id}/providers/Microsoft.Compute/virtualMachines/${each.value}"
  output_blob_uri    = azurerm_storage_blob.run_script_output.url
  error_blob_uri     = azurerm_storage_blob.run_script_error.url
  run_as_password    = random_string.id.result
  run_as_user        = local.vmss_admin_username

  source {
    script_uri = azurerm_storage_blob.run_script_setup.url
    script_uri_managed_identity { ## Block Blob
      client_id = azurerm_user_assigned_identity.environment.client_id
    }
  }

  error_blob_managed_identity { ## Append Blob
    client_id = azurerm_user_assigned_identity.environment.client_id
  }

  output_blob_managed_identity { ## Append Blob
    client_id = azurerm_user_assigned_identity.environment.client_id
  }

  parameter {
    name  = "examplev1"
    value = "val1"
  }

  protected_parameter {
    name  = "examplev2"
    value = "val2"
  }

  tags = { for key, value in module.environment_resource_group.resource.tags : key => value if lower(key) != "created" }
  depends_on = [
    module.virtualmachinescaleset,
    azurerm_role_assignment.run_script_blob_contributor1,
    azurerm_role_assignment.run_script_blob_contributor2,
    azurerm_role_assignment.run_output_script_blob_contributor1,
    azurerm_role_assignment.run_output_script_blob_contributor2,
    azurerm_role_assignment.run_error_script_blob_contributor1,
    azurerm_role_assignment.run_error_script_blob_contributor2
  ]
}
