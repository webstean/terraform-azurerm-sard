resource "azapi_update_resource" "aca_env_otel" {
  type      = "Microsoft.App/managedEnvironments@2024-10-02-preview"
  name      = azurerm_container_app_environment.this.name
  parent_id = azurerm_resource_group.environment.id

  body = {
    properties = {
      appInsightsConfiguration = {
        connectionString = module.application_insights.resource.connection_string
      }

      # This is normal ACA stdout/stderr logging.
      appLogsConfiguration = {
        #destination = "azure-monitor" ## 'azure-monitor', 'log-analytics' or 'none'
        destination = "log-analytics"
        logAnalyticsConfiguration = {
          customerId = module.log_analytics_workspace.resource.workspace_id
          sharedKey  = module.log_analytics_workspace.resource.primary_shared_key
        }
      }

      # This is the managed OpenTelemetry routing.
      openTelemetryConfiguration = {
        tracesConfiguration = {
          destinations = ["appInsights"]
          includeDapr  = false ##tobool(var.enable_dapr)
        }

        logsConfiguration = {
          destinations = ["appInsights"]
        }

        # Do not send metrics to appInsights.
        # Application Insights endpoint supports logs + traces, not metrics.
        metricsConfiguration = {
          destinations = []
        }
      }
    }
    response_export_values = ["*"]
  }

  depends_on = [
    azurerm_container_app_environment.this,
    module.application_insights,
    module.log_analytics_workspace
  ]
}
