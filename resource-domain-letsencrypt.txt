## create certificate for pubsub - expires in 90-day Let's Encrypt expiry, so this GitHub action needs to run every <90 days
resource "acme_certificate" "pubsub" {
  for_each = local.appconfiguration_sku == "premium" ? local.ingress_aliases_pubsub : toset([])

  account_key_pem = acme_registration.this.account_key_pem
  profile         = "tlsserver" ## classic, tlsserver, shortlived
  common_name     = "pubsub.${azurerm_dns_zone.environment.name}"
  #subject_alternative_names = [
  #  "${local.cert_domain}"
  #]
  min_days_remaining            = local.min_days_remaining
  revoke_certificate_on_destroy = true
  dns_challenge {
    provider = "azuredns"
    config = {
      AZURE_AUTH_METHOD     = "oidc"
      AZURE_TENANT_ID       = data.azurerm_client_config.current.tenant_id
      AZURE_CLIENT_ID       = data.azurerm_client_config.current.client_id
      AZURE_SUBSCRIPTION_ID = data.azurerm_client_config.current.subscription_id
      AZURE_RESOURCE_GROUP  = module.environment_resource_group.resource.name
      AZURE_ZONE_NAME       = azurerm_dns_zone.environment.name
      AZURE_TTL             = 300
    }
  }
  depends_on = [
    azurerm_dns_cname_record.pubsub,
  ]
}
## write/update certificate in KeyVault so you can use it in other services
resource "azurerm_key_vault_certificate" "letsencrypt-pubsub" {
  for_each = local.appconfiguration_sku == "premium" ? local.ingress_aliases_pubsub : toset([])

  name         = "cert-${lower(replace(acme_certificate.pubsub[each.key].certificate_domain, ".", "-"))}"
  key_vault_id = module.cert_keyvault.resource_id

  certificate {
    contents = acme_certificate.pubsub[each.key].certificate_p12
  }
}

## create wildcard certificate for aca
resource "acme_certificate" "aca" {
  account_key_pem = acme_registration.this.account_key_pem
  profile         = "tlsserver" ## classic, tlsserver, shortlived
  common_name     = azurerm_dns_zone.aca.name
  subject_alternative_names = [
    "*.${azurerm_dns_zone.aca.name}" ## wildcard SAN - DNS-01 challenge is mandatory for wildcards
  ]
  min_days_remaining            = local.min_days_remaining
  revoke_certificate_on_destroy = true
  dns_challenge {
    provider = "azuredns"
    config = {
      AZURE_AUTH_METHOD     = "oidc"
      AZURE_TENANT_ID       = data.azurerm_client_config.current.tenant_id
      AZURE_CLIENT_ID       = data.azurerm_client_config.current.client_id
      AZURE_SUBSCRIPTION_ID = data.azurerm_client_config.current.subscription_id
      AZURE_RESOURCE_GROUP  = module.environment_resource_group.resource.name
      AZURE_ZONE_NAME       = azurerm_dns_zone.aca.name
      AZURE_TTL             = 300
    }
  }
  depends_on = [
    azurerm_dns_a_record.aca,
    azurerm_dns_txt_record.aca,
    azurerm_dns_zone.aca,
    azurerm_dns_ns_record.aca
  ]
}
## write/update certificate in KeyVault so you can use it in other services
resource "azurerm_key_vault_certificate" "letsencrypt-aca" {
  name         = "cert-wildcard-${lower(replace(acme_certificate.aca.certificate_domain, ".", "-"))}"
  key_vault_id = module.cert_keyvault.resource_id

  certificate {
    contents = acme_certificate.aca.certificate_p12
  }
}
