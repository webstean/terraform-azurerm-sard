resource "azurerm_storage_container" "sandbox_container" {
  count = (tobool(var.data_pii) || tobool(var.data_phi) || tobool(var.deploy_private_endpoints)) ? 0 : 1

  name                  = "sandboxes-containers"
  storage_account_id    = azurerm_storage_account.this.id
  container_access_type = "private"
}
resource "azurerm_role_assignment" "sandbox_container_access" {
  count = (tobool(var.data_pii) || tobool(var.data_phi) || tobool(var.deploy_private_endpoints)) ? 0 : 1

  scope                = azurerm_storage_container.sandbox_container[0].id
  role_definition_name = "Storage Blob Data Owner"
  principal_id         = azurerm_user_assigned_identity.environment.principal_id
}

module "sandbox_group" {
  count = (tobool(var.data_pii) || tobool(var.data_phi) || tobool(var.deploy_private_endpoints)) ? 0 : 1

  source = "git::https://github.com/Azure/terraform-provider-aca.git//modules/sandbox_groups?ref=debaefd1c08d8156a3291c48432dd4ca63e0fbed"

  name              = "sandbox-group-${local.aca_sandbox_name}"
  resource_group_id = module.environment_resource_group.resource_id
  location          = module.environment_resource_group.resource.location
  identity = {
    type         = "UserAssigned"
    identity_ids = [azurerm_user_assigned_identity.environment.id]
  }

  data_plane_operators = {
    terraform_caller = {
      principal_id   = data.azurerm_client_config.current.object_id
      principal_type = "User"
    }
  }

  api_profile             = "rich_preview"
  default_cpu             = "1"
  default_memory          = "2Gi"
  default_disk            = "20Gi"
  max_sandbox_count       = 50
  default_timeout_seconds = 3600

  vnet_connections = {
    default = {
      subnet_id = module.sandbox_subnet[0].resource_id
    }
  }

  tags = { for key, value in module.environment_resource_group.resource.tags : key => value if lower(key) != "created" }
}

data "aca_sandbox_public_disk_image" "ubuntu" {
  sandbox_group_id = module.sandbox_group.id
  location         = module.sandbox_group.location
  name             = "ubuntu"

  depends_on = [module.sandbox_group]
}

resource "aca_sandbox" "ubuntu" {
  sandbox_group_id = module.sandbox_group.id
  location         = module.sandbox_group.location
  name             = "sandbox-group-${local.aca_sandbox_name}-ubuntu"

  source = {
    public_disk_image = data.aca_sandbox_public_disk_image.ubuntu.name
  }

  resources = {
    cpu    = "1000m"
    memory = "2048Mi"
    disk   = "20Gi"
  }

  entrypoint = ["/bin/sh", "-c"]
  command    = ["while true; do sleep 3600; done"]

  auto_suspend = {
    enabled          = true
    interval_seconds = 600
    mode             = "Memory"
  }

  egress_policy = {
    default_action = "Allow"
  }

  labels = {
    quickstart = "sandbox"
  }

  deletion_policy = "Delete"

  timeouts = {
    create = "20m"
    delete = "10m"
  }
}

output "sandbox_group_id" {

}
